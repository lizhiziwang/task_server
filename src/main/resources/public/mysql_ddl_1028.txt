CREATE TABLE `trad_account` (
  `id` bigint NOT NULL,
  `ceate_time` timestamp NOT NULL,
  `update_time` timestamp NOT NULL,
  `des_file` varchar(512) DEFAULT NULL COMMENT '描述文件',
  `des_text` varchar(512) DEFAULT NULL COMMENT '描述文字',
  `price` double DEFAULT NULL COMMENT '期望交易价格',
  `geme_type` varchar(32) DEFAULT NULL COMMENT '游戏类型',
  `game_id` varchar(255) DEFAULT NULL COMMENT '游戏账号',
  `pub_user` bigint DEFAULT NULL COMMENT '发布用户',
  `is_exist` smallint DEFAULT NULL COMMENT '是否存在',
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;
